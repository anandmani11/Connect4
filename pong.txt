#include <cmath>
#include <iostream>
#include <raylib.h>

int player_score = 0;
int cpu_score = 0;
bool game_started = false;

class Ball {
public:
  float x, y;
  int speed_x, speed_y;
  int radius;

  void Draw() { DrawCircle(x, y, radius, Color{249, 214, 91, 255}); }

  void Update() {
    if (!game_started)
      return;

    x += speed_x;
    y += speed_y;

    if (y + radius >= GetScreenHeight() || y - radius <= 0) {
      speed_y *= -1;
    }
  }
};

class Paddle {
protected:
  void LimitMovement() {
    if (y <= 0) {
      y = 0;
    }
    if (y + height >= GetScreenHeight()) {
      y = GetScreenHeight() - height;
    }
  }

public:
  float x, y;
  float width, height;
  int speed;
  std::string direction;

  void Draw() { DrawRectangleRounded({x, y, width, height}, 0.7f, 24, WHITE); }

  void Update() {
    if (IsKeyDown(KEY_UP) || IsKeyDown(KEY_LEFT)) {
      y -= speed;
      direction = "up";
      game_started = true;
    }
    else if (IsKeyDown(KEY_DOWN) || IsKeyDown(KEY_RIGHT)) {
      y += speed;
      direction = "down";
      game_started = true;
    }
    else {
      direction = "still";
    }

    LimitMovement();
  }
};

class CpuPaddle : public Paddle {
private:
  int target_y = 0;
  int reaction_timer = 0;
  const int reaction_delay = 5;
  int error_offset = 0;
  bool coming = false;

public:
  void Update(int ball_x, int ball_y, float ball_speed_x, int screen_width) {
    if (!game_started) {
      direction = "still";
      return;
    }
    coming = (ball_speed_x > 0);

    int current_speed = speed;

    if (!coming) {
      current_speed = speed / 2;
      if (current_speed < 1)
        current_speed = 1;
      target_y = ball_y;
    }
    else {
      reaction_timer++;
      if (reaction_timer >= reaction_delay) {
        error_offset = (rand() % 51) - 25;
        target_y = ball_y + error_offset;
        reaction_timer = 0;
      }
    }

    float paddle_center = y + height / 2;

    if (paddle_center > target_y) {
      y -= current_speed;
      direction = "up";
    }
    else if (paddle_center < target_y) {
      y += current_speed;
      direction = "down";
    }
    else {
      direction = "still";
    }

    LimitMovement();
  }
};

Ball ball;
Paddle player;
CpuPaddle cpu;

void CheckScores() {
  if (ball.x + ball.radius >= GetScreenWidth()) {
    ball.speed_x *= -1;
    player_score++;
    ball.x = GetScreenWidth() / 2;
    ball.y = GetScreenHeight() / 2;
    player.y = GetScreenHeight() / 2 - GetScreenHeight() / 14;
    cpu.y = GetScreenHeight() / 2 - GetScreenHeight() / 14;
    game_started = false;
  }
  if (ball.x - ball.radius <= 0) {
    ball.speed_x *= -1;
    cpu_score++;
    ball.x = GetScreenWidth() / 2;
    ball.y = GetScreenHeight() / 2;
    player.y = GetScreenHeight() / 2 - GetScreenHeight() / 14;
    cpu.y = GetScreenHeight() / 2 - GetScreenHeight() / 14;
    game_started = false;
  }
}

int main() {

  const int screen_width = 960;
  const int screen_height = 600;

  const int paddle_height = screen_height / 7;
  const int paddle_width = screen_width / 55;

  float roundness = 0.5f;
  int segments = 24;

  InitWindow(screen_width, screen_height, "Pong");
  SetTargetFPS(60);

  ball.radius = 10;
  ball.x = screen_width / 2;
  ball.y = screen_height / 2;
  ball.speed_x = 7;
  ball.speed_y = 7;

  player.x = 20;
  player.y = screen_height / 2 - paddle_height / 2;
  player.width = paddle_width;
  player.height = paddle_height;
  player.speed = 7;
  player.direction = "still";

  cpu.x = screen_width - (20 + paddle_width);
  cpu.y = screen_height / 2 - paddle_height / 2;
  cpu.width = paddle_width;
  cpu.height = paddle_height;
  cpu.speed = 7;
  cpu.direction = "still";

  while (WindowShouldClose() == false) {
    BeginDrawing();

    // update
    ball.Update();
    player.Update();
    cpu.Update(ball.x, ball.y, ball.speed_x, screen_width);
    CheckScores();

    // Collision
    if (CheckCollisionCircleRec(
            Vector2{ball.x, ball.y}, ball.radius,
            Rectangle{player.x, player.y, player.width, player.height})) {

      ball.x = player.x + player.width + ball.radius;

      if (player.direction == "up") {
        ball.speed_x = 9;
      }
      else if (player.direction == "still") {
        ball.speed_x = 7;
      }
      else if (player.direction == "down") {
        ball.speed_x = 5;
      }
    }

    if (CheckCollisionCircleRec(
            Vector2{ball.x, ball.y}, ball.radius,
            Rectangle{cpu.x, cpu.y, cpu.width, cpu.height})) {

      ball.x = cpu.x - ball.radius;

      if (cpu.direction == "up") {
        ball.speed_x = -9;
      }
      else if (cpu.direction == "still") {
        ball.speed_x = -7;
      }
      else if (cpu.direction == "down") {
        ball.speed_x = -5;
      }
    }

    // Drawing
    ClearBackground(BLACK);
    DrawRectangle(screen_width / 2, 0, screen_width / 2, screen_height,
                  Color{40, 186, 155, 255});
    DrawRectangle(0, 0, screen_width / 2, screen_height,
                  Color{20, 160, 133, 255});
    DrawCircle(screen_width / 2, screen_height / 2, screen_height / 6,
               Color{255, 255, 255, 100});

    player.Draw();
    cpu.Draw();

    DrawLine(screen_width / 2, 0, screen_width / 2, screen_height, WHITE);
    ball.Draw();

    int fontSize = 80;

    const char *cpu_text = TextFormat("%i", cpu_score);
    int cpu_width = MeasureText(cpu_text, fontSize);
    DrawText(cpu_text, (screen_width / 4) - (cpu_width / 2), 20, fontSize,
             WHITE);

    const char *player_text = TextFormat("%i", player_score);
    int player_width = MeasureText(player_text, fontSize);
    DrawText(player_text, ((screen_width / 4) * 3) - (player_width / 2), 20,
             fontSize, WHITE);

    if (!game_started) {
      const char *start_text = "Press Up/Down to Start";
      int text_fontSize = 20;
      int text_width = MeasureText(start_text, text_fontSize);

      DrawText(start_text, (screen_width / 2) - (text_width / 2),
               screen_height - 40, text_fontSize, GRAY);
    }

    EndDrawing();
  }

  CloseWindow();
  return 0;
}
